cmake_minimum_required(VERSION 3.20)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ArcanumMPPolishSetup VERSION 1.0.0 LANGUAGES CXX)
    enable_testing()
endif()
set(ARCMP_POLISH_EXTRACTOR "" CACHE FILEPATH "innoextract binary to embed for this target")
option(ARCMP_POLISH_DESKTOP_UI "Use statically linked SDL for Linux desktop feedback" OFF)
if(ARCMP_POLISH_EXTRACTOR)
    file(READ "${ARCMP_POLISH_EXTRACTOR}" ARCMP_POLISH_EXTRACTOR_HEX HEX)
    string(REGEX REPLACE "(..)" "0x\\1," ARCMP_POLISH_EXTRACTOR_BYTES "${ARCMP_POLISH_EXTRACTOR_HEX}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ARCMP_POLISH_EXTRACTOR}")
endif()
configure_file(extractor_data.h.in generated/extractor_data.h @ONLY)
add_library(arcmp_polish_import STATIC import.cpp)
target_compile_features(arcmp_polish_import PUBLIC cxx_std_17)
add_executable(arcmp-polish-setup main.cpp)
target_link_libraries(arcmp-polish-setup PRIVATE arcmp_polish_import)
target_include_directories(arcmp-polish-setup PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(arcmp-polish-setup PROPERTIES OUTPUT_NAME ArcanumMP-Polski)
if(WIN32)
    target_link_libraries(arcmp-polish-setup PRIVATE user32)
    if(MINGW)
        target_link_options(arcmp-polish-setup PRIVATE -municode -static-libstdc++ -static-libgcc)
    endif()
elseif(ARCMP_POLISH_DESKTOP_UI)
    if(NOT TARGET SDL3::SDL3-static)
        find_package(SDL3 REQUIRED CONFIG COMPONENTS SDL3-static)
    endif()
    target_link_libraries(arcmp-polish-setup PRIVATE SDL3::SDL3-static)
    target_compile_definitions(arcmp-polish-setup PRIVATE ARCMP_POLISH_SDL=1)
endif()
if(NOT MSVC)
    target_compile_options(arcmp_polish_import PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(arcmp-polish-setup PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(arcmp-polish-setup PRIVATE "$<$<CONFIG:Release>:-s>")
    if(NOT WIN32)
        target_link_options(arcmp-polish-setup PRIVATE -static-libstdc++ -static-libgcc)
    endif()
else()
    target_compile_options(arcmp-polish-setup PRIVATE /utf-8)
endif()
add_executable(polish-import-test test.cpp)
target_link_libraries(polish-import-test PRIVATE arcmp_polish_import)
if(MINGW)
    target_link_options(polish-import-test PRIVATE -static-libstdc++ -static-libgcc)
endif()
add_test(NAME polish-import COMMAND polish-import-test)
